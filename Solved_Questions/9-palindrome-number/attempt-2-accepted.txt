class Solution {
public:
    bool isPalindrome(int x) {
        if(x<0 || (x%10==0 && x!=0 ))
        return false;
        int rev_half_num=0;
        while(x>rev_half_num){
            rev_half_num = rev_half_num*10 + x%10;
            x/=10;
        }
        return x == rev_half_num || x == rev_half_num / 10;
        
    }
};
